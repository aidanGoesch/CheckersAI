function simulateRandomCheckersGame():
    board = initializeBoard()
    currentPlayer = RED
    
    while not isGameOver(board):
        moves = getQuickRandomMove(board, currentPlayer)
        if moves is null:
            break
        
        applyMove(board, moves)
        
        while isJumpMove(moves) and canContinueJumping(board, moves.endPosition):
            additionalJump = getQuickRandomJump(board, moves.endPosition)
            if additionalJump is null:
                break
            applyMove(board, additionalJump)
            moves = additionalJump
        
        currentPlayer = oppositePlayer(currentPlayer)
    
    return getGameResult(board)
